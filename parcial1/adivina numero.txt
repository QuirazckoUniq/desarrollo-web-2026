const numeroSecreto = Math.floor(Math.random() * 100) + 1;
let intentos = 0;
let adivinado = false;

while (!adivinado) {
  let intento = parseInt(prompt("Adivina el número (1-100):"));
  intentos++;

  if (intento > numeroSecreto) {
    alert("Muy alto");
  } else if (intento < numeroSecreto) {
    alert("Muy bajo");
  } else {
    alert("¡Correcto! Lo lograste en " + intentos + " intentos.");
    adivinado = true;
  }
}
